git clone клонирует удаленный репозиторий на ваг пк
git status показывает состояние рабочей директории
git add добавляет изменение в индекс
git commit -m создает комент с добавленными в индекс изменениями
git push отправляет локальные коммиты в удаленный репозиторий
git pull загружает изменения из удаленного репозитория и сливает их с текущей веткой
git branch показывает список веток или создает новую
git checkout переглючаеться на другую ветку
git merge сливает указанную ветку в текущию
git log показывает историю коммитов текущей ветки
git diff показывает различия между рабочими файлами и последним коммитом
git init создает новый локальный репозиторий в текущей папке
